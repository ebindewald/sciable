# RNA-FM (per-residue embeddings) Runpod Serverless worker dependencies.
#
# Upstream code (MIT): https://github.com/ml4bio/RNA-FM — the pip/importable package is
# `rna-fm` (imported as `fm`). We install the pinned PyPI RELEASE, NOT from git.
#
# ⚠ Why PyPI, not git: the repo's setup.py does `open("README_backup.md")`, but that file
# is NOT in the repo — so `pip install git+https://github.com/ml4bio/RNA-FM@<commit>` dies
# during metadata generation (this is what failed the first Runpod build, 2026-11-08). The
# PyPI release ships a py3-none-any WHEEL, which pip installs WITHOUT running setup.py, so
# the bug can't fire. rna-fm==0.2.2 is an immutable, reproducible pin.
#
# rna-fm pulls its own install_requires: numpy, pandas, tqdm, scikit-learn, ptflops.
# ⚠ ptflops is load-bearing: `import fm` runs fm/__init__.py -> `from .downstream import *`
#   -> baseline.py, which has an UNCONDITIONAL `from ptflops import ...`. We never use it,
#   but the package won't import without it. (apex is try/except-guarded; torchvision is
#   only imported inside a function — neither needed.)
#
# ⚠ torch is NOT pinned/listed: the image builds on
# pytorch/pytorch:2.7.1-cuda12.8-cudnn9-runtime (modern-GPU kernels, sm_50..sm_120).
# Upstream's environment.yml pins torch 1.9/py3.8, but the model is a plain BERT encoder
# (nn.Module + F.*), so it should run on 2.7 — confirm on the first smoke run. rna-fm's
# install_requires does NOT list torch, so pip will not replace the base image's build.

runpod>=1.6

rna-fm==0.2.2
